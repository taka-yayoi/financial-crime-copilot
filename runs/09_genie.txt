SQL generated by Genie:
WITH `rule_fp` AS (SELECT `rule_code`, SUM(`alerts`) AS `total_alerts`, SUM(`suspicious`) AS `total_suspicious`, SUM(`alerts`) - SUM(`suspicious`) AS `false_positive_count`, ROUND(100.0 * (1 - try_divide(SUM(`suspicious`), SUM(`alerts`))), 1) AS `false_positive_pct` FROM `serverless_stable_unstructured_catalog`.`aml_gold`.`alert_kpis` GROUP BY `rule_code`), `ranked_rules` AS (SELECT `rule_code`, `total_alerts`, `total_suspicious`, `false_positive_count`, `false_positive_pct`, RANK() OVER (ORDER BY `false_positive_count` DESC) AS `fp_rank` FROM `rule_fp`) SELECT `rule_code`, `total_alerts`, `total_suspicious`, `false_positive_count`, `false_positive_pct` FROM `ranked_rules` ORDER BY `fp_rank`, `rule_code`
